const userRef = db.collection('users').doc(userId);
const userDoc = await userRef.get();

if (!userDoc.exists) {
  return res.status(404).json({ success: false, message: "User not found" });
}

const userData = userDoc.data();

// Check if replacement already used
if (userData.replacementUsed) {
  return res.status(400).json({ 
    success: false, 
    message: "Replacement lock active! Once replacement/cancellation opportunity is used, it cannot be used again." 
  });
}

// Process replacement and permanently lock it
await userRef.update({
  replacementUsed: true,
  replacementOrderId: orderId,
  replacementDate: admin.firestore.FieldValue.serverTimestamp()
});

return res.status(200).json({ 
  success: true, 
  message: "Replacement processed successfully. Your replacement option is now permanently locked." 
});
